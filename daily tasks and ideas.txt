Step 1 — Fix Core Correctness

Before anything else, fix:

A) Merge frequency overwrite bug

Current:

new_vocab[tuple(new_word)] = freq

Correct:

new_vocab[tuple(new_word)] = (
    new_vocab.get(tuple(new_word), 0) + freq
)
B) Add unknown token handling

Right now unknown tokens crash silently conceptually.

Add:

def tokens_to_ids(tokens, token_to_id):
    unk_id = token_to_id["<unk>"]

    return [
        token_to_id.get(token, unk_id)
        for token in tokens
    ]

def decode(tokens):

And:

def ids_to_tokens(ids, id_to_token):
    return [
        id_to_token[i]
        for i in ids
    ]

Then full roundtrip:

tokens = encode("lowest", merges)

ids = tokens_to_ids(tokens, token_to_id)

recovered_tokens = ids_to_tokens(ids, id_to_token)

decoded = decode(recovered_tokens)

def save_tokenizer(merges, token_to_id, path="tokenizer"):
    import os

    os.makedirs(path, exist_ok=True)

    with open(f"{path}/merges.txt", "w") as f:
        for pair in merges:
            f.write(f"{pair[0]} {pair[1]}\n")

    with open(f"{path}/vocab.json", "w") as f:
        json.dump(token_to_id, f, indent=2)
Load merges
def load_tokenizer(path="tokenizer"):
    import json

    merges = []

    with open(f"{path}/merges.txt") as f:
        for line in f:
            a, b = line.strip().split()
            merges.append((a, b))

    with open(f"{path}/vocab.json") as f:
        token_to_id = json.load(f)

    id_to_token = {
        int(v): k
        for k, v in token_to_id.items()
    }

    return merges, token_to_id, id_to_token

Example metric
def avg_tokens_per_word(corpus, merges):
    total = 0

    for word in corpus:
        total += len(encode(word, merges))

    return total / len(corpus)

Step 7 — Text → IDs

Build:

def tokenize_text(text, merges, token_to_id):
    words = text.split()

    ids = []

    for word in words:
        tokens = encode(word, merges)
        ids.extend(tokens_to_ids(tokens, token_to_id))

    return ids

Now you have:

text → token ids

Exactly like GPT.
